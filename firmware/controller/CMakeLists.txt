cmake_minimum_required(VERSION 3.20)
project(robomaster_supercap_firmware LANGUAGES C ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(FW_MODE "SAFE_MONITOR_ONLY" CACHE STRING "SAFE_MONITOR_ONLY, LOW_ENERGY_COMMISSIONING, POWER_STAGE_CONTROL")
set_property(CACHE FW_MODE PROPERTY STRINGS SAFE_MONITOR_ONLY LOW_ENERGY_COMMISSIONING POWER_STAGE_CONTROL)
if(NOT FW_MODE MATCHES "^(SAFE_MONITOR_ONLY|LOW_ENERGY_COMMISSIONING|POWER_STAGE_CONTROL)$")
 message(FATAL_ERROR "Invalid FW_MODE")
endif()
set(APP_SOURCES App/sensing.c App/state_machine.c App/control.c App/protocol.c App/bq76942.c config/rev_a_config.c config/calibration_defaults.c config/bq_profile.c)
if(CMAKE_CROSSCOMPILING)
 foreach(dep HAL_ROOT DEVICE_ROOT CMSIS_ROOT)
  file(TO_CMAKE_PATH "${${dep}}" ${dep})
  if(NOT IS_DIRECTORY "${${dep}}")
   message(FATAL_ERROR "Set ${dep}; see Drivers/README.md. No automatic download/install.")
  endif()
 endforeach()
 set(HAL_NAMES hal hal_adc hal_adc_ex hal_cortex hal_dma hal_dma_ex hal_exti hal_fdcan hal_flash hal_flash_ex hal_gpio hal_hrtim hal_iwdg hal_pwr hal_pwr_ex hal_rcc hal_rcc_ex hal_spi hal_uart hal_uart_ex)
 set(HAL_SOURCES)
 foreach(name IN LISTS HAL_NAMES)
  list(APPEND HAL_SOURCES "${HAL_ROOT}/Src/stm32g4xx_${name}.c")
 endforeach()
 add_library(vendor STATIC ${HAL_SOURCES} "${DEVICE_ROOT}/Source/Templates/system_stm32g4xx.c")
 target_include_directories(vendor PUBLIC Core/Inc "${HAL_ROOT}/Inc" "${DEVICE_ROOT}/Include" "${CMSIS_ROOT}/CMSIS/Core/Include")
 target_compile_definitions(vendor PUBLIC STM32G474xx USE_HAL_DRIVER)
 target_compile_options(vendor PUBLIC -mcpu=cortex-m4 -mthumb -mfpu=fpv4-sp-d16 -mfloat-abi=hard -ffunction-sections -fdata-sections)
 target_compile_options(vendor PRIVATE -Os -Wall -Wextra)
 add_executable(controller.elf ${APP_SOURCES} Core/Src/board.c Core/Src/main.c Core/Src/interrupts.c Core/Src/syscalls.c "${DEVICE_ROOT}/Source/Templates/gcc/startup_stm32g474xx.s")
 target_include_directories(controller.elf PRIVATE App)
 target_compile_definitions(controller.elf PRIVATE FW_MODE=${FW_MODE})
 target_compile_options(controller.elf PRIVATE -Os -g3 -Wall -Wextra -Werror -Wshadow -Wstrict-prototypes -Wdouble-promotion)
 target_link_libraries(controller.elf PRIVATE vendor m)
 target_link_options(controller.elf PRIVATE -mcpu=cortex-m4 -mthumb -mfpu=fpv4-sp-d16 -mfloat-abi=hard --specs=nano.specs --specs=nosys.specs "-T${CMAKE_CURRENT_SOURCE_DIR}/STM32G474RE_FLASH.ld" -Wl,--gc-sections "-Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/controller.map")
 add_custom_command(TARGET controller.elf POST_BUILD COMMAND "${CMAKE_SIZE}" $<TARGET_FILE:controller.elf> COMMAND "${CMAKE_OBJCOPY}" -O ihex $<TARGET_FILE:controller.elf> "${CMAKE_CURRENT_BINARY_DIR}/controller.hex")
else()
 enable_testing()
 foreach(mode SAFE_MONITOR_ONLY LOW_ENERGY_COMMISSIONING POWER_STAGE_CONTROL)
  add_executable(test_${mode} ${APP_SOURCES} tests/test_firmware.c)
  target_include_directories(test_${mode} PRIVATE App)
  target_compile_definitions(test_${mode} PRIVATE FW_MODE=${mode})
  target_compile_options(test_${mode} PRIVATE -Wall -Wextra -Werror -Wshadow -Wstrict-prototypes -Wdouble-promotion -g)
  target_link_libraries(test_${mode} PRIVATE m)
  add_test(NAME ${mode} COMMAND test_${mode})
  set_tests_properties(${mode} PROPERTIES TIMEOUT 10)
 endforeach()
endif()
